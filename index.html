// ============================================================================
// GOOGLE APPS SCRIPT BACKEND UNTUK TASK FLOW
// (STRICT EXISTING SCHEMA, FAST 1-STEP LOGIN & PERSISTENT SESSION TOKEN)
// ============================================================================

const SHEET_NAMES = {
  TASKS: 'Tasks',
  KATEGORI: 'Kategori',
  PRIORITY: 'Priority',
  PIC: 'PIC',
  STATUS: 'Status',
  REFERENCE: 'Reference',
  ACTION_PLAN: 'ActionPlan'
};

const SCHEMAS = {
  [SHEET_NAMES.TASKS]: [
    'ID', 'Tanggal', 'Kategori', 'Task', 'Priority', 'PIC',
    'TargetDate', 'Status', 'Progress', 'Reference', 'ActionPlan', 'Catatan'
  ],
  [SHEET_NAMES.KATEGORI]: ['Kode', 'Nama', 'Group', 'Keterangan'],
  [SHEET_NAMES.PRIORITY]: ['Kode', 'Nama', 'SLA_Hari'],
  [SHEET_NAMES.PIC]: ['Kode', 'Nama', 'Initial', 'Divisi', 'Role', 'Username', 'Password'],
  [SHEET_NAMES.STATUS]: ['Kode', 'Nama', 'Warna', 'Completed'],
  [SHEET_NAMES.REFERENCE]: ['Kode', 'Nama', 'Keterangan'],
  [SHEET_NAMES.ACTION_PLAN]: ['Kode', 'Nama', 'KategoriTerkait']
};

const SESSION_TTL_SECONDS = 21600; // Masa aktif sesi: 6 jam
const SESSION_TTL_MS = SESSION_TTL_SECONDS * 1000;

function createJsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(
    ContentService.MimeType.JSON
  );
}

/**
 * Mengambil sheet yang SUDAH ADA saja. Tidak pernah membuat sheet atau kolom baru.
 */
function getExistingSheet(ss, sheetName) {
  const sheet = ss.getSheetByName(sheetName);
  if (!sheet || sheet.getLastColumn() === 0) return null;
  return sheet;
}

function createSessionToken(userObj) {
  const token =
    Utilities.getUuid().replace(/-/g, '') + Utilities.getUuid().replace(/-/g, '');
  const expiresAt = Date.now() + SESSION_TTL_MS;
  const sessionPayload = JSON.stringify({
    user: userObj,
    expiresAt: expiresAt
  });

  try {
    CacheService.getScriptCache().put('TF_SESSION_' + token, sessionPayload, SESSION_TTL_SECONDS);
  } catch (e) {}

  try {
    PropertiesService.getScriptProperties().setProperty('TF_SESSION_' + token, sessionPayload);
  } catch (e) {}

  cleanupExpiredSessions();
  return token;
}

function validateSessionToken(token) {
  const cleanToken = String(token || '').trim();
  if (!cleanToken) return null;

  const key = 'TF_SESSION_' + cleanToken;
  const cache = CacheService.getScriptCache();
  let raw = cache.get(key);

  if (!raw) {
    try {
      raw = PropertiesService.getScriptProperties().getProperty(key);
    } catch (e) {
      raw = null;
    }
  }

  if (!raw) return null;

  try {
    const parsed = JSON.parse(raw);
    const user = parsed.user || parsed;
    const expiresAt = Number(parsed.expiresAt || 0);

    if (expiresAt > 0 && Date.now() > expiresAt) {
      revokeSessionToken(cleanToken);
      return null;
    }

    const renewedPayload = JSON.stringify({
      user: user,
      expiresAt: Date.now() + SESSION_TTL_MS
    });

    try { cache.put(key, renewedPayload, SESSION_TTL_SECONDS); } catch (e) {}
    try { PropertiesService.getScriptProperties().setProperty(key, renewedPayload); } catch (e) {}

    return user;
  } catch (err) {
    return null;
  }
}

function revokeSessionToken(token) {
  const cleanToken = String(token || '').trim();
  if (!cleanToken) return;
  const key = 'TF_SESSION_' + cleanToken;

  try { CacheService.getScriptCache().remove(key); } catch (e) {}
  try { PropertiesService.getScriptProperties().deleteProperty(key); } catch (e) {}
}

function cleanupExpiredSessions() {
  try {
    const props = PropertiesService.getScriptProperties();
    const all = props.getProperties();
    const now = Date.now();

    Object.keys(all).forEach(k => {
      if (!k.startsWith('TF_SESSION_')) return;
      try {
        const parsed = JSON.parse(all[k]);
        if (parsed.expiresAt && now > Number(parsed.expiresAt)) {
          props.deleteProperty(k);
        }
      } catch (err) {
        props.deleteProperty(k);
      }
    });
  } catch (e) {}
}

function sanitizeSheetRows(sheetName, rows, sessionUser) {
  if (sheetName === SHEET_NAMES.PIC) {
    return rows.map(item => {
      const { Password, password, ...safeRow } = item;
      return safeRow;
    });
  }

  if (
    sheetName === SHEET_NAMES.TASKS &&
    sessionUser &&
    String(sessionUser.Role || '').trim().toLowerCase() === 'staff'
  ) {
    const targetName = String(sessionUser.Nama || '').trim().toLowerCase();
    return rows.filter(t => String(t.PIC || '').trim().toLowerCase() === targetName);
  }

  return rows;
}

function fetchAuthorizedDatabase(sessionUser) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const result = {};

  Object.keys(SCHEMAS).forEach(name => {
    const sheet = getExistingSheet(ss, name);
    if (!sheet) {
      result[name] = [];
      return;
    }

    const expectedHeaders = SCHEMAS[name];
    const data = sheet.getDataRange().getValues();

    if (data.length <= 1) {
      result[name] = [];
      return;
    }

    const actualHeaders = data[0].map(h => String(h).trim());
    const mappedRows = data
      .slice(1)
      .filter(row => row.some(cell => cell !== '' && cell !== null))
      .map(row => {
        const obj = {};
        actualHeaders.forEach((h, idx) => {
          if (!h) return;
          let val = row[idx];
          if (val instanceof Date) {
            val = Utilities.formatDate(val, 'GMT+7', 'yyyy-MM-dd');
          }
          obj[h] = val;
        });

        expectedHeaders.forEach(stdKey => {
          const matchedKey = Object.keys(obj).find(k => k.toLowerCase() === stdKey.toLowerCase());
          let val = matchedKey !== undefined ? obj[matchedKey] : '';
          if (val instanceof Date) {
            val = Utilities.formatDate(val, 'GMT+7', 'yyyy-MM-dd');
          }
          obj[stdKey] = val;
        });

        return obj;
      });

    result[name] = sanitizeSheetRows(name, mappedRows, sessionUser);
  });

  return result;
}

function authenticatePic(username, password) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = getExistingSheet(ss, SHEET_NAMES.PIC);
  if (!sheet) {
    return { status: 'error', message: 'Sheet PIC tidak ditemukan di Spreadsheet.' };
  }

  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) {
    return { status: 'error', message: 'Data pada sheet PIC masih kosong.' };
  }

  const headers = data[0].map(h => String(h).trim());
  const userIdx = headers.findIndex(h => h.toLowerCase() === 'username');
  const passIdx = headers.findIndex(h => h.toLowerCase() === 'password');

  if (userIdx === -1 || passIdx === -1) {
    return { status: 'error', message: 'Kolom Username atau Password belum tersedia pada sheet PIC.' };
  }

  const inputUser = String(username || '').trim().toLowerCase();
  const inputPass = String(password || '').trim();

  if (!inputUser || !inputPass) {
    return { status: 'error', message: 'Username dan Password wajib diisi.' };
  }

  const expectedHeaders = SCHEMAS[SHEET_NAMES.PIC];

  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    const dbUser = String(row[userIdx] || '').trim().toLowerCase();
    const dbPass = String(row[passIdx] || '').trim();

    if (dbUser && dbUser === inputUser && dbPass === inputPass) {
      const userObj = {};
      expectedHeaders.forEach(stdKey => {
        if (stdKey.toLowerCase() === 'password') return;
        const colIdx = headers.findIndex(h => h.toLowerCase() === stdKey.toLowerCase());
        let val = colIdx !== -1 ? row[colIdx] : '';
        if (val instanceof Date) {
          val = Utilities.formatDate(val, 'GMT+7', 'yyyy-MM-dd');
        }
        userObj[stdKey] = val;
      });

      const token = createSessionToken(userObj);
      const workspaceData = fetchAuthorizedDatabase(userObj);
      return { status: 'success', token: token, user: userObj, data: workspaceData };
    }
  }

  return { status: 'error', message: 'Username atau Password salah.' };
}

function doGet(e) {
  const action = String((e && e.parameter && e.parameter.action) || '').trim().toUpperCase();
  if (action === 'PING') {
    return createJsonResponse({ status: 'connected', app: 'Task Flow' });
  }
  return createJsonResponse({
    status: 'unauthorized',
    message: 'Akses ditolak. Silakan login melalui aplikasi Task Flow.'
  });
}

function doPost(e) {
  const payload = JSON.parse((e && e.postData && e.postData.contents) || '{}');
  const action = String(payload.action || '').trim().toUpperCase();

  if (action === 'PING') return createJsonResponse({ status: 'connected', app: 'Task Flow' });
  if (action === 'LOGIN') return createJsonResponse(authenticatePic(payload.username, payload.password));
  if (action === 'LOGOUT') {
    revokeSessionToken(payload.token);
    return createJsonResponse({ status: 'success' });
  }

  const sessionUser = validateSessionToken(payload.token);
  if (!sessionUser) {
    return createJsonResponse({
      status: 'unauthorized',
      message: 'Sesi login tidak valid atau telah berakhir. Silakan login kembali.'
    });
  }

  if (action === 'GET_DATA') {
    const data = fetchAuthorizedDatabase(sessionUser);
    return createJsonResponse({ status: 'success', user: sessionUser, data: data });
  }

  const lock = LockService.getScriptLock();
  lock.tryLock(15000);

  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const userRole = String(sessionUser.Role || 'Staff').trim().toLowerCase();

    if (action === 'SAVE_TASK') {
      const sheet = getExistingSheet(ss, SHEET_NAMES.TASKS);
      if (!sheet) return createJsonResponse({ status: 'error', message: 'Sheet Tasks tidak ditemukan.' });

      const task = payload.data || {};
      if (userRole === 'staff') task.PIC = sessionUser.Nama;

      const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim());
      const rows = sheet.getDataRange().getValues();

      const rowValues = headers.map(h => {
        const key = Object.keys(task).find(k => k.toLowerCase() === h.toLowerCase());
        let val = key !== undefined ? task[key] : '';
        if ((h.toLowerCase() === 'tanggal' || h.toLowerCase() === 'targetdate') && /^\d{4}-\d{2}-\d{2}$/.test(String(val))) {
          const parts = String(val).split('-');
          val = new Date(Number(parts[0]), Number(parts[1]) - 1, Number(parts[2]));
        }
        return val;
      });

      const idColIdx = headers.findIndex(h => h.toLowerCase() === 'id');
      let foundRow = -1;
      if (idColIdx !== -1) {
        for (let i = 1; i < rows.length; i++) {
          if (String(rows[i][idColIdx]) === String(task.ID)) {
            foundRow = i + 1;
            break;
          }
        }
      }

      if (foundRow > 0) {
        sheet.getRange(foundRow, 1, 1, rowValues.length).setValues([rowValues]);
        applyDateFormatting(sheet, headers, foundRow, 1);
      } else {
        sheet.appendRow(rowValues);
        applyDateFormatting(sheet, headers, sheet.getLastRow(), 1);
      }
    } else if (action === 'DELETE_TASK') {
      const sheet = getExistingSheet(ss, SHEET_NAMES.TASKS);
      if (!sheet) return createJsonResponse({ status: 'error', message: 'Sheet Tasks tidak ditemukan.' });

      const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim());
      const idColIdx = headers.findIndex(h => h.toLowerCase() === 'id');
      const picColIdx = headers.findIndex(h => h.toLowerCase() === 'pic');
      const rows = sheet.getDataRange().getValues();

      if (idColIdx !== -1) {
        for (let i = 1; i < rows.length; i++) {
          if (String(rows[i][idColIdx]) === String(payload.id)) {
            if (userRole === 'staff' && picColIdx !== -1) {
              if (String(rows[i][picColIdx] || '').trim().toLowerCase() !== String(sessionUser.Nama || '').trim().toLowerCase()) {
                return createJsonResponse({ status: 'error', message: 'Anda tidak berhak menghapus task milik PIC lain.' });
              }
            }
            sheet.deleteRow(i + 1);
            break;
          }
        }
      }
    } else if (action === 'SYNC_ALL') {
      if (userRole !== 'admin' && userRole !== 'manager') {
        return createJsonResponse({ status: 'error', message: 'Hanya Admin atau Manager yang dapat memperbarui Master Data.' });
      }

      const db = payload.database || {};
      Object.keys(db).forEach(sheetName => {
        if (sheetName === SHEET_NAMES.TASKS) return;
        const sheet = getExistingSheet(ss, sheetName);
        if (!sheet) return;

        const items = db[sheetName] || [];
        const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim());

        let existingPassMap = {};
        if (sheetName === SHEET_NAMES.PIC && sheet.getLastRow() > 1) {
          const oldData = sheet.getDataRange().getValues();
          const kodeIdx = headers.findIndex(h => h.toLowerCase() === 'kode');
          const userIdx = headers.findIndex(h => h.toLowerCase() === 'username');
          const passIdx = headers.findIndex(h => h.toLowerCase() === 'password');
          if (passIdx !== -1) {
            for (let r = 1; r < oldData.length; r++) {
              const kVal = kodeIdx !== -1 ? String(oldData[r][kodeIdx] || '').trim().toLowerCase() : '';
              const uVal = userIdx !== -1 ? String(oldData[r][userIdx] || '').trim().toLowerCase() : '';
              const pVal = oldData[r][passIdx];
              if (kVal) existingPassMap['kode:' + kVal] = pVal;
              if (uVal) existingPassMap['user:' + uVal] = pVal;
            }
          }
        }

        if (sheet.getLastRow() > 1) {
          sheet.getRange(2, 1, sheet.getLastRow() - 1, headers.length).clearContent();
        }

        if (items && items.length > 0) {
          const rows = items.map(item =>
            headers.map(h => {
              const key = Object.keys(item).find(k => k.toLowerCase() === h.toLowerCase());
              let val = key !== undefined && item[key] !== undefined ? item[key] : '';
              if (sheetName === SHEET_NAMES.PIC && h.toLowerCase() === 'password' && !String(val).trim()) {
                const kLookup = String(item.Kode || '').trim().toLowerCase();
                const uLookup = String(item.Username || '').trim().toLowerCase();
                val = existingPassMap['kode:' + kLookup] ?? existingPassMap['user:' + uLookup] ?? '';
              }
              return val;
            })
          );
          sheet.getRange(2, 1, rows.length, headers.length).setValues(rows);
        }
      });
    }

    return createJsonResponse({ status: 'success' });
  } finally {
    lock.releaseLock();
  }
}

function applyDateFormatting(sheet, headers, startRow, numRows) {
  headers.forEach((h, idx) => {
    if (h.toLowerCase() === 'tanggal' || h.toLowerCase() === 'targetdate') {
      sheet.getRange(startRow, idx + 1, numRows, 1).setNumberFormat('dd/MM/yyyy');
    }
  });
}
