Koneksi Database Google Spreadsheet

Hubungkan aplikasi web ini langsung dengan file Google Spreadsheet Anda menggunakan Google Apps Script Web App API.

Kode Backend Google Apps Script (Code.gs)

Tempel di Extensions > Apps Script pada Google Sheet Anda → Deploy > Manage deployments > New version (Akses: Anyone). 

// ============================================================================
// GOOGLE APPS SCRIPT BACKEND UNTUK TASK FLOW (VERSI 2.4 - FULL SCHEMA SYNC)
// ============================================================================

const SHEET_NAMES = {
  TASKS: 'Tasks',
  KATEGORI: 'Kategori',
  PRIORITY: 'Priority',
  PIC: 'PIC',
  STATUS: 'Status',
  REFERENCE: 'Reference',
  ACTION_PLAN: 'ActionPlan'
};

const SCHEMAS = {
  [SHEET_NAMES.TASKS]: ['ID', 'Tanggal', 'Kategori', 'Task', 'Priority', 'PIC', 'TargetDate', 'Status', 'Progress', 'Reference', 'ActionPlan', 'Catatan'],
  [SHEET_NAMES.KATEGORI]: ['Kode', 'Nama', 'Group', 'Keterangan'],
  [SHEET_NAMES.PRIORITY]: ['Kode', 'Nama', 'SLA_Hari'],
  [SHEET_NAMES.PIC]: ['Kode', 'Nama', 'Initial', 'Divisi', 'Role', 'Username', 'Password'],
  [SHEET_NAMES.STATUS]: ['Kode', 'Nama', 'Warna', 'Completed'],
  [SHEET_NAMES.REFERENCE]: ['Kode', 'Nama', 'Keterangan'],
  [SHEET_NAMES.ACTION_PLAN]: ['Kode', 'Nama', 'KategoriTerkait']
};

function setupSheets() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  Object.keys(SCHEMAS).forEach(sheetName => {
    const expectedHeaders = SCHEMAS[sheetName];
    let sheet = ss.getSheetByName(sheetName);

    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      sheet.appendRow(expectedHeaders);
      formatHeaderRow(sheet, expectedHeaders.length);
    } else {
      const lastCol = sheet.getLastColumn();
      if (lastCol === 0) {
        sheet.appendRow(expectedHeaders);
        formatHeaderRow(sheet, expectedHeaders.length);
      } else {
        const existingHeaders = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(h => String(h).trim());
        expectedHeaders.forEach(colName => {
          const exists = existingHeaders.some(eh => eh.toLowerCase() === colName.toLowerCase());
          if (!exists) {
            const newColIdx = sheet.getLastColumn() + 1;
            sheet.getRange(1, newColIdx).setValue(colName);
            existingHeaders.push(colName);
          }
        });
        formatHeaderRow(sheet, sheet.getLastColumn());
      }
    }
  });
}

function formatHeaderRow(sheet, numCols) {
  const headerRange = sheet.getRange(1, 1, 1, numCols);
  headerRange.setFontWeight('bold')
             .setBackground('#1e293b')
             .setFontColor('#ffffff')
             .setVerticalAlignment('middle');
  sheet.setFrozenRows(1);
}

function doGet(e) {
  setupSheets();
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const result = {};

  Object.keys(SCHEMAS).forEach(name => {
    const sheet = ss.getSheetByName(name);
    const expectedHeaders = SCHEMAS[name];
    const data = sheet.getDataRange().getValues();

    if (data.length <= 1) {
      result[name] = [];
      return;
    }

    const actualHeaders = data[0].map(h => String(h).trim());
    result[name] = data.slice(1)
      .filter(row => row.some(cell => cell !== '' && cell !== null))
      .map(row => {
        let obj = {};
        actualHeaders.forEach((h, idx) => {
          if (!h) return;
          let val = row[idx];
          if (val instanceof Date) {
            val = Utilities.formatDate(val, 'GMT+7', 'yyyy-MM-dd');
          }
          obj[h] = val;
        });
        expectedHeaders.forEach(stdKey => {
          const matchedKey = Object.keys(obj).find(k => k.toLowerCase() === stdKey.toLowerCase());
          let val = matchedKey !== undefined ? obj[matchedKey] : '';
          if (val instanceof Date) {
            val = Utilities.formatDate(val, 'GMT+7', 'yyyy-MM-dd');
          }
          obj[stdKey] = val;
        });
        return obj;
      });
  });

  return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: result }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.tryLock(15000);

  try {
    setupSheets();
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const payload = JSON.parse(e.postData.contents);
    const action = payload.action;

    if (action === 'SAVE_TASK') {
      const sheet = ss.getSheetByName(SHEET_NAMES.TASKS);
      const task = payload.data;
      const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim());
      const rows = sheet.getDataRange().getValues();

      const rowValues = headers.map(h => {
        const key = Object.keys(task).find(k => k.toLowerCase() === h.toLowerCase());
        let val = key !== undefined ? task[key] : '';
        if ((h.toLowerCase() === 'tanggal' || h.toLowerCase() === 'targetdate') && /^\d{4}-\d{2}-\d{2}$/.test(String(val))) {
          const parts = String(val).split('-');
          val = new Date(Number(parts[0]), Number(parts[1]) - 1, Number(parts[2]));
        }
        return val;
      });

      const idColIdx = headers.findIndex(h => h.toLowerCase() === 'id');
      let foundRow = -1;
      if (idColIdx !== -1) {
        for (let i = 1; i < rows.length; i++) {
          if (String(rows[i][idColIdx]) === String(task.ID)) {
            foundRow = i + 1;
            break;
          }
        }
      }

      if (foundRow > 0) {
        sheet.getRange(foundRow, 1, 1, rowValues.length).setValues([rowValues]);
        applyDateFormatting(sheet, headers, foundRow, 1);
      } else {
        sheet.appendRow(rowValues);
        applyDateFormatting(sheet, headers, sheet.getLastRow(), 1);
      }
    }
    else if (action === 'DELETE_TASK') {
      const sheet = ss.getSheetByName(SHEET_NAMES.TASKS);
      const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim());
      const idColIdx = headers.findIndex(h => h.toLowerCase() === 'id');
      const rows = sheet.getDataRange().getValues();

      if (idColIdx !== -1) {
        for (let i = 1; i < rows.length; i++) {
          if (String(rows[i][idColIdx]) === String(payload.id)) {
            sheet.deleteRow(i + 1);
            break;
          }
        }
      }
    }
    else if (action === 'SYNC_ALL') {
      const db = payload.database;
      Object.keys(db).forEach(sheetName => {
        const sheet = ss.getSheetByName(sheetName);
        if (!sheet) return;
        const items = db[sheetName];
        const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => String(h).trim());

        if (sheet.getLastRow() > 1) {
          sheet.getRange(2, 1, sheet.getLastRow() - 1, headers.length).clearContent();
        }
        if (items && items.length > 0) {
          const rows = items.map(item => headers.map(h => {
            const key = Object.keys(item).find(k => k.toLowerCase() === h.toLowerCase());
            let val = key !== undefined && item[key] !== undefined ? item[key] : '';
            if (sheetName === SHEET_NAMES.TASKS && (h.toLowerCase() === 'tanggal' || h.toLowerCase() === 'targetdate') && /^\d{4}-\d{2}-\d{2}$/.test(String(val))) {
              const parts = String(val).split('-');
              val = new Date(Number(parts[0]), Number(parts[1]) - 1, Number(parts[2]));
            }
            return val;
          }));
          sheet.getRange(2, 1, rows.length, headers.length).setValues(rows);
          if (sheetName === SHEET_NAMES.TASKS) {
            applyDateFormatting(sheet, headers, 2, rows.length);
          }
        }
      });
    }

    return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function applyDateFormatting(sheet, headers, startRow, numRows) {
  headers.forEach((h, idx) => {
    if (h.toLowerCase() === 'tanggal' || h.toLowerCase() === 'targetdate') {
      sheet.getRange(startRow, idx + 1, numRows, 1).setNumberFormat('dd/MM/yyyy');
    }
  });
}
